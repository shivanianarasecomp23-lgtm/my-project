ML- sales data clustering

# Step 1: Import libraries
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans


# Step 2: Load dataset
df = pd.read_csv("sales_data_sample.csv", encoding='latin-1')

print(df.head())
print(df.shape)


# Step 3: Select numerical columns
X = df[['QUANTITYORDERED', 'PRICEEACH', 'SALES',
        'QTR_ID', 'MONTH_ID', 'YEAR_ID', 'MSRP']]


# Step 4: Scale data
sc = StandardScaler()

X = sc.fit_transform(X)


# Step 5: Elbow Method
wcss = []

for i in range(1, 11):
    kmeans = KMeans(n_clusters=i, random_state=42, n_init=10)
    kmeans.fit(X)
    wcss.append(kmeans.inertia_)

plt.plot(range(1, 11), wcss, marker='o')

plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.title("Elbow Method")

plt.show()


# Step 6: Apply K-Means
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)

y_pred = kmeans.fit_predict(X)

print("Cluster Labels:")
print(y_pred)

print("Number of Clusters:", len(set(y_pred)))


# Step 7: Display cluster counts
print(pd.Series(y_pred).value_counts().sort_index())
